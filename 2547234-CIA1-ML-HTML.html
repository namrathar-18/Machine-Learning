def iqr_outliers(data, col):
    Q1 = data[col].quantile(0.25)
    Q3 = data[col].quantile(0.75)
    IQR = Q3 - Q1
    lower = Q1 - 1.5*IQR
    upper = Q3 + 1.5*IQR
    outliers = data[(data[col] < lower) | (data[col] > upper)]
    return len(outliers), lower, upper

cols_to_check = ['daily_social_media_hours', 'sleep_hours', 'stress_level', 'anxiety_level']
outlier_summary = []
for c in cols_to_check:
    n_out, low, up = iqr_outliers(df, c)
    outlier_summary.append([c, low, up, n_out, round(100*n_out/df[c].notna().sum(),2)])

outlier_df = pd.DataFrame(outlier_summary, columns=["Variable", "Lower Bound", "Upper Bound", "Outlier Count", "Outlier %"])
